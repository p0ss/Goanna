[Goanna client options]

#    Let a Goanna client draw terrain past this server's send distance: from
#    mapblocks it has already been sent, and from coarse summaries this mod
#    gives of terrain the server has already generated, which is how places
#    the player has never been are drawn. Nothing is generated to answer a
#    summary, and what is drawn from the client's own store is as old as the
#    last time the player was near it.
goanna_far_rendering (Far rendering) bool false

#    How far past the send distance, in nodes. Costs the client memory and
#    draw calls, and bounds the far terrain summaries below.
goanna_far_rendering_distance (Far rendering distance) int 256 64 8192

#    Optional horizon for a registered procedural surface provider. Unlike
#    the ordinary distance above, this does not widen pregeneration or map
#    reads: the provider returns only coarse synthetic columns. Leave it at
#    the ordinary distance unless the provider has cheap persistent coverage.
goanna_far_provider_distance (Procedural far-provider distance) int 256 64 8192

#    How many mapblocks this mod may read from the map each server step to
#    summarise them. Summaries are what let a client draw places its player
#    has never been: the server reads terrain it has already generated (it
#    never generates any for this), keeps a coarse record of each block, and
#    answers a client's request from those records. A block is read once,
#    when it is generated, when it changes, or the first time it is wanted
#    on a world older than the store, so this paces the reading, not the
#    answering: an area the store knows is answered at once. Lower it on a
#    busy server; 0 does not disable the feature, goanna_far_rendering does.
goanna_far_summary_blocks_per_step (Far summary blocks per step) int 96 1 512

#    Read the areas around each player into the summary store in the
#    background, nearest first, when nothing else is waiting. On a world
#    generated under this mod there is nothing for it to do; on an older one
#    it is how the map gets read once instead of on the first client's clock.
goanna_far_summary_backfill (Far summary backfill) bool true

#    How many 8 by 8 by 8 mapblock areas of summaries to keep in memory.
#    About 11 KB each. Past this, areas untouched for ten minutes are written
#    to mod storage and let go, and come back when next wanted.
goanna_far_summary_cache_areas (Far summary areas in memory) int 4096 64 65536

#    Stop reading summaries while the server's step length, smoothed over
#    about a second, is above this, in seconds, and carry on where the part finished area left off
#    once it drops. The same guard and the same default as pregeneration's,
#    and the reason the rate above can be raised at all.
goanna_far_summary_lag (Far summary lag limit) float 0.5 0.0 60.0

#    Generate terrain around each player out to the far rendering distance,
#    one 128 node area at a time, a slice of an area per emerge call, and
#    hand each area's summary to the clients near it. A summary can only
#    describe terrain that exists, and a server generates only within the
#    range a client asks for, so without this a new world has a 192 node
#    horizon whatever the distance above says. Costs mapgen time and map
#    memory for terrain no one has
#    visited, which is why it is off here and on for the server a Goanna
#    client launches for itself.
goanna_far_pregenerate (Far pregeneration) bool false

#    Seconds to wait between one pregenerated area and the next. The pause
#    between the slices within an area is nothing: an area is emerged a slice
#    at a time so it never fills the queue the player's own blocks come
#    through, and that is what keeps it out of the way, not this.
goanna_far_pregenerate_interval (Far pregeneration interval) float 1.0 0.0 600.0

#    Mapblocks on a side of one pregeneration emerge call. The whole point of
#    slicing is that the server's emerge queue is a FIFO and Lua's emerge
#    ignores its limits, so a large slice is a long wait for the player's own
#    blocks behind it. Rounded down to 1, 2, 4 or 8.
goanna_far_pregenerate_slice (Far pregeneration slice) int 4 1 8

#    Number of 128-node areas emerged concurrently. Each stream has at most
#    one slice in the emerge queue, so this raises throughput without one
#    enormous forced batch blocking the player's live terrain. Reduce to 1
#    on a CPU-limited public server.
goanna_far_pregenerate_concurrency (Far pregeneration concurrency) int 2 1 8

#    Write far-pipeline progress, queue depths and Lua memory to the server
#    log every 30 seconds. Useful for diagnosing long-running generation.
goanna_far_log_stats (Log far pipeline statistics) bool false

#    Seconds of server step time above which pregeneration waits for the
#    server to catch up, against the step length smoothed over about a
#    second. A value near the step length would still read as "behind" on a
#    busy server; half a second means something is really wrong.
goanna_far_pregenerate_lag (Far pregeneration lag limit) float 0.5 0.0 60.0

#    Source engine style movement: air strafing, bunny hopping, surfing. Needs
#    disable_anticheat as well, or the server rejects the movement it produces.
goanna_source_movement (Source style movement) bool false

goanna_source_movement_air_accel (Air acceleration) float 100.0 1.0 1000.0

goanna_source_movement_max_air_speed (Max air speed) float 3.0 0.1 100.0

#    Shared dig damage. A Goanna client already carves the block it is digging
#    on its own screen, and needs nothing from a server to do that. With this
#    on it also reports the carve, the server stores it on the node, and OTHER
#    players see the same worn block.
#
#    That last step is the one thing here a client cannot be taken at its word
#    for: a modified client could report a shape it never dug. Reports are
#    limited to blocks within reach, capped in size and rate limited, but they
#    are still the client's account. Nothing about the world's contents
#    changes, and this is not dig progress: how long a block takes to break is
#    a check the server enforces either way.
#
#    Reasonable for a world you host for yourself and people you know; not for
#    a public server, which is why it is off.
goanna_shared_dig_damage (Shared dig damage) bool false

#    Remember how far each block has been dug. A block a player digs part of
#    the way and leaves keeps its progress, so coming back to it takes only
#    the rest of its dig time, for any player. The server measures this
#    itself from the player holding dig and pointing at the block, against
#    the block's own dig time for the tool held, and finishes the dig through
#    the block's own dig rules; nothing a client reports is trusted. A dig
#    done in one go is unchanged. Progress is kept until the block changes or
#    the server stops. Worlds started from Goanna turn it on.
goanna_dig_progress (Remember dig progress) bool false

[Director]

#    The director: a language model acting as game master, connected by the
#    operator (docs/director.md in the Goanna repository). Enabling this only
#    makes it available. Nothing happens until a director process answers at
#    the url in <world>/goanna_director.conf, which also holds the token; that
#    file is created on first start and is never broadcast. Needs this mod in
#    secure.http_mods. Players are told on joining that a director is active
#    and can opt out with /director optout.
goanna_director (Director) bool false

#    How the director service reaches this server. file: small files in
#    <world>/goanna_director/link, for a service on this machine; needs nothing
#    else. http: the service's HTTP endpoint at the url in
#    <world>/goanna_director.conf, for a service elsewhere; needs a Luanti
#    built with curl and goanna_server_mod in secure.http_mods.
goanna_director_transport (Director transport) enum file file,http

#    Encounter points the director may spend per hour, across all players. A
#    mob costs its maximum health over 10 plus its damage: a Mineclonia
#    zombie 5, a skeleton 4.
goanna_director_points_per_hour (Director encounter points per hour) int 40 0 10000

#    Creatures the director may have alive at once, in all and per player.
goanna_director_max_entities (Director creatures in all) int 24 0 256
goanna_director_max_entities_per_player (Director creatures per player) int 8 0 64

#    Lines a minute a director character (or the narrator) may say, and lines
#    a minute any one player may be sent. The longest line, in characters.
goanna_director_speech_per_minute (Director lines per speaker per minute) int 6 0 120
goanna_director_listener_per_minute (Director lines per listener per minute) int 12 0 240
goanna_director_speech_chars (Director line length) int 280 20 1000

#    How far a character's voice carries, in nodes. A character speaks only to
#    players this close, and hears lines addressed to it only from them.
goanna_director_earshot (Director earshot) float 24 4 128

#    Encounters are sized to each player's gear: the largest is this base
#    plus the player's gear score (0 to 1, armour and best hotbar weapon
#    against the game's best) times the second number.
goanna_director_encounter_base (Director encounter base) float 4 0 100
goanna_director_encounter_per_gear (Director encounter per gear) float 16 0 200

#    Seconds before a staged encounter's remaining mobs are removed.
goanna_director_leash (Director encounter leash) int 180 10 600

#    Seconds an encounter queued for the next build up may wait.
goanna_director_queue_s (Director queue time) int 300 10 3600

#    Where staged mobs may stand. standable: loaded, clear, on walkable ground,
#    unprotected. natural: also the mob framework's own spawn rules (light,
#    for one), which refuse most monsters in daylight.
goanna_director_spawn_rules (Director spawn rules) enum standable standable,natural

#    No director mob is placed within this many nodes of static spawn, nor
#    within this (or 32, the larger) of a player who opted out.
goanna_director_exclusion_radius (Director exclusion radius) int 24 0 256

#    Mobs the director may never create. Creepers and endermen change the
#    world, which the director may not.
goanna_director_deny (Director deny list) string mobs_mc:creeper,mobs_mc:enderman,mobs_mc:wither,mobs_mc:enderdragon,mobs_mc:ghast,mobs_mc:ravager,mobs_mc:evoker

#    What the director sees of players. exact: positions, health and gear,
#    which the operator can see anyway. coarse: the region and a gear score.
#    Players are told which in the notice on joining.
goanna_director_sees (Director sees) enum exact exact,coarse

#    Which public chat the director reads. all: every public line, as the
#    operator can. addressed: only lines that start with a character's name,
#    from players within earshot of it. none: nothing. Private messages and
#    chat commands never reach any mod.
goanna_director_chat (Director reads chat) enum all addressed,all,none

#    Whether characters may remember short lines the model writes, as well as
#    facts recorded from events, and how many and how long. Every one is in
#    the audit log. Memory lasts for the server session.
goanna_director_memory_text (Director memory lines) bool true
goanna_director_memory_lines (Director memory lines per pair) int 8 0 32
goanna_director_memory_chars (Director memory line length) int 120 10 200

#    Pacing: the intensity that ends a build up, the seconds of relax after a
#    fade and after a death, and the relax a player gets on joining.
goanna_director_peak (Director peak intensity) float 0.7 0.1 1.0
goanna_director_relax (Director relax) float 45 0 600
goanna_director_relax_after_death (Director relax after death) float 120 0 1200
goanna_director_join_grace (Director join grace) float 30 0 600

#    Rewards: points per hour for items the director makes (an item's value
#    plus each enchantment's level), separate from encounter points; the
#    most one reward may cost; and items never granted, comma separated.
goanna_director_reward_points_per_hour (Director reward points per hour) int 30 0 10000
goanna_director_reward_max (Director largest reward) int 12 0 1000
goanna_director_reward_deny (Director reward deny list) string

#    Structures: whether the director may put buildings in the world at all,
#    nodes per hour it may change, the largest bounding box of one, and how
#    many nodes a second a character lays when ordered to build. A building
#    is never placed on protected ground or where players have dug or built,
#    and undo puts the ground back.
goanna_director_structures (Director structures) bool false
goanna_director_build_nodes_per_hour (Director build nodes per hour) int 4000 0 1000000
goanna_director_structure_max_volume (Director largest structure) int 32768 1 262144
goanna_director_build_rate (Director build rate) float 4 0.5 20

#    Where players can read more about this server's director, named in the
#    notice on joining. Optional.
goanna_director_info (Director information) string
